{{- /* head/fonts-local.html Emits local @font-face declarations for the Google Fonts vendored in assets/fonts/ (the raw CSS lives in assets/css/fonts-local.css). assets/ files are not published automatically, so each font file is fingerprinted (which renders it to the output as /fonts/..woff2) and the matching /fonts/ URL in the CSS is rewritten to the hashed URL. The raw CSS and the font files are never modified; the URLs are stable across builds because they are content hashes. Arg (dict): fonts — list of font file names in assets/fonts/ (the $fontFiles list from head.html, which has already verified each file exists) */ -}} {{- $css := resources.Get "css/fonts-local.css" }} {{- if not $css }} {{- errorf "vendor/fonts: missing local font CSS assets/css/fonts-local.css" }} {{- end }} {{- $cssText := $css.Content }} {{- range $f := .fonts }} {{- with resources.Get (printf "fonts/%s" $f) }} {{- $published := . | fingerprint }} {{- $cssText = strings.Replace $cssText (printf "/fonts/%s" $f) $published.RelPermalink }} {{- else }} {{- errorf "vendor/fonts: missing local font file %q" $f }} {{- end }} {{- end }}
